cmake_minimum_required(VERSION 3.24)
project(CrossShot VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CROSSSHOT_BUILD_GUI "Build Qt probes" ON)
option(CROSSSHOT_BUILD_OCR "Build native ONNX OCR worker" ON)
include(CTest)
add_library(crossshot_core libs/core/core.cpp)
target_include_directories(crossshot_core PUBLIC libs/core)
if(BUILD_TESTING)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE crossshot_core)
 add_test(NAME core_contracts COMMAND core_tests)
endif()
if(CROSSSHOT_BUILD_GUI)
 find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Test)
 set(CMAKE_AUTOMOC ON)
 add_library(crossshot_widgets apps/desktop/pin.cpp apps/desktop/fixture.cpp)
 target_include_directories(crossshot_widgets PUBLIC apps/desktop)
 target_link_libraries(crossshot_widgets PUBLIC Qt6::Widgets crossshot_core)
 if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  add_library(crossshot_platform platform/macos/capture.mm)
  target_compile_options(crossshot_platform PRIVATE -fobjc-arc)
  target_link_libraries(crossshot_platform PUBLIC Qt6::Widgets "-framework ScreenCaptureKit" "-framework AppKit" "-framework CoreGraphics")
 else()
  add_library(crossshot_platform platform/stub.cpp)
  target_link_libraries(crossshot_platform PUBLIC Qt6::Widgets)
 endif()
 target_include_directories(crossshot_platform PUBLIC platform)
 qt_add_executable(CrossShot MACOSX_BUNDLE apps/desktop/main.cpp)
 target_link_libraries(CrossShot PRIVATE crossshot_widgets crossshot_platform Qt6::Test)
 set_target_properties(CrossShot PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.crossshot.p0 MACOSX_BUNDLE_BUNDLE_NAME CrossShot MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in")
 if(BUILD_TESTING)
  qt_add_executable(widget_tests tests/widget_tests.cpp)
  target_link_libraries(widget_tests PRIVATE crossshot_widgets crossshot_platform Qt6::Test)
  add_test(NAME widget_contracts COMMAND widget_tests -platform offscreen)
 endif()
 if(CROSSSHOT_BUILD_OCR)
  set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted ONNX Runtime SDK")
  find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
  find_library(ORT_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
  qt_add_executable(crossshot-ocr providers/paddle/main.cpp providers/paddle/ocr.cpp)
  target_include_directories(crossshot-ocr PRIVATE "${ORT_INCLUDE}" providers/paddle)
  target_link_libraries(crossshot-ocr PRIVATE Qt6::Gui crossshot_core "${ORT_LIBRARY}")
  if(APPLE)
   add_custom_target(bundle_ocr ALL COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:crossshot-ocr> "$<TARGET_BUNDLE_DIR:CrossShot>/Contents/MacOS/crossshot-ocr" DEPENDS CrossShot crossshot-ocr)
   add_dependencies(CrossShot crossshot-ocr)
  endif()
 endif()
 install(TARGETS CrossShot BUNDLE DESTINATION . RUNTIME DESTINATION bin)
endif()
